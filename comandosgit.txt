# ==============================
# GIT INTERMEDIÁRIO - COMANDOS
# ==============================

# ---------- BRANCHES ----------

# Lista as branches locais (a atual aparece com *)
git branch

# Muda para uma branch existente
git switch bug-fix

# Cria uma branch (continua na branch atual)
git branch bug-fix

# Cria a branch e já muda para ela
git switch -c bug-fix

# Renomeia uma branch
git branch -m bug-fix bug-fix-responsividade

# Exclui uma branch já mesclada
git branch -d bug-fix

# Força a exclusão (mesmo sem merge, cuidado: pode perder commits)
git branch -D bug-fix


# ---------- DIFF ----------

# Diferenças de todos os arquivos não preparados vs. último commit
git diff

# Diferenças de um arquivo não preparado vs. último commit
git diff index.html

# Diferenças de todos os arquivos preparados vs. último commit
git diff --staged

# Diferenças de um arquivo preparado vs. último commit
git diff --staged index.html

# Diferenças entre dois commits (usando hashes)
git diff 35f95fe 85fes9e

# Diferenças entre dois commits (usando HEAD)
git diff HEAD~1 HEAD~3

# Diferenças entre duas branches
git diff main bug-fix


# ---------- MERGE ----------

# Vai para a branch de destino e traz o trabalho da outra
git switch main
git merge bug-fix


# ---------- REPOSITÓRIOS REMOTOS ----------

# Clona um repositório (já configura o remoto como origin)
git clone https://github.com/usuario/projeto.git

# Vincula um repositório local a um remoto
git remote add origin https://github.com/usuario/projeto.git

# Baixa as novidades do remoto SEM mesclar
git fetch origin

# Baixa e mescla as novidades (fetch + merge)
git pull origin main

# Envia seus commits para o remoto
git push origin main

# Primeira vez que envia uma branch nova (vincula ao remoto)
git push -u origin bug-fix

# Compara sua main com a main do remoto (após o fetch)
git diff main origin/main
